cmake_minimum_required(VERSION 3.16)
project(sol_analyzer_native CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(solagg SHARED aggregator.cpp)
target_compile_options(solagg PRIVATE -Wall -Wextra -O2)
add_executable(solagg_test test.cpp)
target_link_libraries(solagg_test solagg)
enable_testing()
add_test(NAME solagg_test COMMAND solagg_test)
